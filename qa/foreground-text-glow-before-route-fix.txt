// Selector matching below is a deliberately limited, fail-closed AST model.
// It tests target scoping and simulated hover ancestry, NOT layout, native form
// rendering, actual hit testing, CSS cascade or a browser's computed styles.
function parseSelectors(selector) {
  const result = [];
  transform({ filename: "selector.css", code: Buffer.from(`${selector}{color:green}`),
    visitor: { Selector(value) { result.push(value); } } });
  return result;
}
const selectors = parseSelectors(rules[0].selector);
const oldHoverRules = [];
postcss.parse(css.slice(0, css.indexOf(marker))).walkRules(rule => {
  if (rule.selector?.includes(":hover") && values(rule)["text-shadow"] && rule.selector.includes(":not(:disabled")) {
    oldHoverRules.push(...parseSelectors(rule.selector));
  }
});
const node = (tag, attrs = {}, text = "", children = []) => {
  const value = { tag, attrs, text, children, parent: null, hovered: false };
  children.forEach(child => { child.parent = value; });
  return value;
};
const descendants = element => element.children.flatMap(child => [child, ...descendants(child)]);
const ancestors = element => element?.parent ? [element.parent, ...ancestors(element.parent)] : [];
function hovered(element) {
  const tree = ancestors(element).at(-1) ?? element;
  for (const item of [tree, ...descendants(tree)]) item.hovered = false;
  for (const item of [element, ...ancestors(element)]) item.hovered = true;
}
function matches(element, tokens, scope = null) {
  let split = -1;
  tokens.forEach((token, index) => { if (token.type === "combinator") split = index; });
  const compound = tokens.slice(split + 1);
  if (!compound.every(token => component(element, token))) return false;
  if (split === -1) return true;
  const previous = tokens.slice(0, split);
  const relation = tokens[split].value;
  const candidates = relation === "child" ? [element.parent].filter(Boolean)
    : relation === "descendant" ? ancestors(element) : assert.fail(`Unsupported combinator: ${relation}`);
  return previous.length ? candidates.some(parent => matches(parent, previous, scope)) : candidates.includes(scope);
}
function component(element, token) {
  if (token.type === "universal") return true;
  if (token.type === "type") return element.tag === token.name;
  if (token.type === "class") return (element.attrs.class ?? "").split(/\s+/).includes(token.name);
  if (token.type === "attribute") {
    if (!Object.hasOwn(element.attrs, token.name)) return false;
    if (!token.operation) return true;
    assert.equal(token.operation.operator, "equal");
    return element.attrs[token.name] === token.operation.value;
  }
  assert.equal(token.type, "pseudo-class", `Unsupported selector token: ${JSON.stringify(token)}`);
  if (token.kind === "root") return !element.parent;
  if (token.kind === "hover") return element.hovered;
  if (token.kind === "disabled") return Object.hasOwn(element.attrs, "disabled");
  if (["active", "focus-visible"].includes(token.kind)) return false;
  if (["is", "where"].includes(token.kind)) return token.selectors.some(selector => matches(element, selector));
  if (token.kind === "not") return !token.selectors.some(selector => matches(element, selector));
  if (token.kind === "has") return token.selectors.some(selector => {
    const relative = selector[0]?.type === "combinator" ? selector : [{ type: "combinator", value: "descendant" }, ...selector];
    return descendants(element).some(child => matches(child, relative, element));
  });
  assert.fail(`Unsupported pseudo-class: ${token.kind}`);
}
const targets = element => selectors.some(selector => matches(element, selector));
const existingControl = element => oldHoverRules.some(selector => matches(element, selector));
function mount(...children) {
  const content = node("div", { class: "aiwork-cinematic-content" }, "", children);
  node("html", {}, "", [node("body", {}, "", [content])]);
  return content;
}

function parseHtml(html) {
  const document = node("document");
  let current = document;
  const voidTags = new Set(["area", "base", "br", "col", "embed", "hr", "img", "input", "link", "meta", "param", "source", "track", "wbr"]);
  // Static React output escapes attribute delimiters. Scripts/styles are not copy.
  const source = html.replace(/<(script|style)\b[^>]*>[\s\S]*?<\/\1>/gi, "").replace(/<!--[\s\S]*?-->/g, "");
  for (const match of source.matchAll(/<(\/?)([\w:-]+)\b([^>]*?)>|([^<]+)/g)) {
    if (match[4]) { current.text += match[4]; continue; }
    const [, closing, tag, attributes] = match;
    if (closing) {
      const found = [current, ...ancestors(current)].find(item => item.tag === tag);
      if (found?.parent) current = found.parent;
      continue;
    }
    const attrs = {};
    for (const attr of attributes.matchAll(/([^\s=/]+)(?:="([^"]*)"|='([^']*)'|=([^\s>]+))?/g)) {
      attrs[attr[1]] = attr[2] ?? attr[3] ?? attr[4] ?? "";
    }
    const child = node(tag, attrs);
    child.parent = current; current.children.push(child);
    if (!voidTags.has(tag)) current = child;
  }
  const htmlNode = document.children.find(item => item.tag === "html");
  assert.ok(htmlNode, "Static route must contain an html root");
  htmlNode.parent = null;
  return htmlNode;
}

// The saved baseline is local QA evidence, not a second runtime implementation.
test("v9 foreground layer stays additive; unrelated v8 sources and assets are unchanged", () => {
  assert.equal(hash(css.slice(0, css.indexOf(marker)).trimEnd() + "\n"), baseline["app/cinematic-theme.css"]);
  assert.equal(baseline["app/cinematic-theme.css"], "76c1ff7b1431667f916b74fbdb01c1c8e0370dbcffb9a26d4787e612cb6bc60a");
  assert.equal(baseline["app/components/matrix-engine.js"], "07b5333383c473ea9fe7f499dc6092fd2311d13a4335c4dd6192bc8fbbefc4f3");
  for (const [file, expected] of Object.entries(baseline)) {
    // v10 scene cleanup is independently guarded by scene-cleanup.test.mjs.
    // v12 restores only card-transition composition; verify the unchanged
    // historical prefix instead of silently replacing the baseline hash.
    if (file === "app/header-flow.css") {
      assert.equal(hash(read(file).split("/* V12 motion repair:")[0].trimEnd() + "\n"), expected, file);
      continue;
    }
    if (![...changedRuntime, "app/cinematic-theme.css", "app/components/matrix-engine.js", "app/components/CinematicMatrix.tsx"].includes(file)) assert.equal(hash(file.endsWith(".css") ? read(file) : fs.readFileSync(path.join(root, file))), expected, file);
  }
  const runtime = ["app", "public"].flatMap(dir => walkFiles(path.join(root, dir))).map(file => path.relative(root, file)).filter(file => !addedRuntime.includes(file)).sort();
  assert.deepEqual(runtime, Object.keys(baseline).sort(), "No new JS, wrappers, handlers, controls or assets");
});

test("new text paint compiles and changes only static color/shadow plus two contrast backings", () => {
  assert.equal(rules.length, 4);
  const compiled = transform({ filename: "cinematic-theme.css", code: Buffer.from(css), errorRecovery: false });
  assert.deepEqual(compiled.warnings, []);
  for (const rule of rules) {
    assert.equal(rule.parent.name, "media");
    assert.equal(rule.parent.params, "(hover: hover) and (pointer: fine)");
    assert.match(rule.selector, /:root body \.aiwork-cinematic-content/);
    for (const [property] of Object.entries(values(rule))) assert.ok(["color", "text-shadow", "background"].includes(property), property);
    for (const declaration of rule.nodes.filter(n => n.type === "decl")) assert.equal(declaration.important, undefined);
  }
  assert.equal(rules.filter(rule => values(rule).background).length, 2);
  assert.doesNotMatch(additions, /@keyframes|will-change\s*:|cursor\s*:|outline\s*:|filter\s*:|animation\s*:|transition\s*:/);
});

test("hovering nested links/code selects that fragment, not its whole paragraph or siblings", () => {
  const a = node("a", { href: "/privacy" }, "Privacy");
  const code = node("code", {}, "drive.file");
  const p = node("p", {}, "Read ", [a, code]);
  const sibling = node("p", {}, "Other paragraph");
  const section = node("section", {}, "", [p, sibling]); mount(section);
  hovered(a);
  assert.ok(targets(a)); assert.ok(!targets(p)); assert.ok(!targets(code)); assert.ok(!targets(sibling)); assert.ok(!targets(section));
  hovered(code); assert.ok(targets(code)); assert.ok(!targets(a)); assert.ok(!targets(p));
  hovered(p); assert.ok(targets(p), "The paragraph's own copy is still a text region");
});

test("foreground-only scope excludes scene glyphs even if the scene is moved inside content", () => {
  const glyph = node("span", {}, "AIWORK 01");
  const scene = node("div", { class: "aiwork-cinematic-scene" }, "", [glyph, node("canvas")]);
  const copy = node("p", {}, "Foreground"); const content = mount(copy);
  scene.parent = content.parent; content.parent.children.unshift(scene);
  hovered(glyph); assert.ok(!targets(glyph)); assert.ok(!targets(copy));

test("prompt and native file button use AAA-contrast backing without changing normal surfaces", () => {
  const luminance = hex => hex.slice(1).match(/../g).map(value => parseInt(value, 16) / 255)
    .map(value => value <= .04045 ? value / 12.92 : ((value + .055) / 1.055) ** 2.4)
    .reduce((sum, value, i) => sum + value * [.2126, .7152, .0722][i], 0);
  assert.ok((luminance("#b8ffcd") + .05) / (luminance("#001c09") + .05) > 15);
  assert.match(rules[1].selector, /\.home-user-prompt/);
  assert.match(rules[3].selector, /::file-selector-button$/);
  assert.match(rules[2].selector, /::placeholder$/);
  for (const rule of rules.filter(rule => values(rule).background)) assert.equal(values(rule).background, "#001c09");
});

test("all exported foreground text parents have an own-hover rule, including non-heading divs", () => {
  const routes = walkFiles(path.join(root, "out")).filter(file => /(?:^|\/)index\.html$/.test(file));
  assert.ok(routes.length >= 17, "Run npm run build:static before these route-coverage checks");
  let checked = 0;
  const uncovered = [];
  for (const route of routes) {
    const document = parseHtml(fs.readFileSync(route, "utf8"));
    for (const copy of descendants(document).filter(item => item.text.trim() && [item, ...ancestors(item)].some(parent => parent.attrs.class?.split(/\s+/).includes("aiwork-cinematic-content")))) {
      const lineage = [copy, ...ancestors(copy)];
      if (lineage.some(item => ["head","script","style","title"].includes(item.tag) || Object.hasOwn(item.attrs, "hidden") || Object.hasOwn(item.attrs, "inert") || Object.hasOwn(item.attrs, "disabled") || item.attrs["aria-disabled"] === "true" || item.attrs.class === "honeypot")) continue;
      hovered(copy);
      checked++;
      if (!targets(copy) && !existingControl(copy)) uncovered.push({ route: path.relative(root, route), tag: copy.tag, class: copy.attrs.class, text: copy.text.trim().slice(0, 50) });
    }
  }
  assert.deepEqual(uncovered, []);
  assert.ok(checked > 1900, `${checked} direct text parents checked across ${routes.length} routes`);
  console.log(`Foreground selector coverage: ${checked} direct text parents across ${routes.length} static route files.`);
});
